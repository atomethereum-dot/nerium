<script>
/* ══ que el traductor del navegador entienda la página ══
   Los titulares —y el párrafo de la cita— se parten en palabras para animar
   su entrada, y el de portada se parte además en letras. Los traductores del
   navegador (el de Chrome y el de Google) traducen nodo de texto a nodo de
   texto: con la frase troceada traducen cada palabra por su cuenta y
   devuelven algo que no llega a ser una frase, y lo que va letra a letra no
   lo traducen en absoluto.

   Aquí se guarda la frase entera de cada trozo nada más partirse, y en cuanto
   se detecta que la página está traducida se devuelve de una pieza para que
   el traductor la lea como lo que es. Se pierde la animación de entrada de
   esos titulares, que es justo lo que sobra cuando alguien está leyendo la
   página en otro idioma. */
(function(){
  /* los mismos que trocea la animación, ya marcados con .sp */
  const PARTIDOS='h1.sp, h2.sp, .say p.sp, .rows b.sp, .duo h3.sp, .start b.sp, .card b.sp';
  const original=new WeakMap();

  /* el troceo por letras marca los espacios con <i class="sp"> vacíos:
     textContent devolvería las palabras pegadas unas a otras */
  function frase(el){
    let t;
    if(el.querySelector(':scope > i.sp')){
      t=[...el.childNodes].map(n=>
        (n.nodeType===1&&n.tagName==='I'&&n.classList.contains('sp'))?' ':n.textContent
      ).join('');
    }else{
      t=el.textContent;
    }
    return t.replace(/\s+/g,' ').trim();
  }

  function guardar(){
    document.querySelectorAll(PARTIDOS).forEach(el=>{
      if(original.has(el))return;
      const t=frase(el);
      if(t)original.set(el,t);
    });
  }

  /* El selector de idioma no lo debe tocar el traductor del navegador: los
     nombres de idioma se leen en su propia lengua, y además su botón es lo que
     se consulta más abajo para saber en qué idioma está la página. El menú se
     construye al abrirlo, así que se vuelve a marcar cada poco. */
  function noTraducir(){
    document.querySelectorAll('.lang, .lang-menu').forEach(e=>{
      if(e.getAttribute('translate')!=='no')e.setAttribute('translate','no');
    });
  }
  noTraducir();
  /* y en cuanto se abre el menú, sin esperar al reloj */
  document.addEventListener('click',e=>{
    if(e.target.closest&&e.target.closest('.lang'))setTimeout(noTraducir,0);
  },true);

  let devuelto=false;
  const vigias=[];
  function entero(){
    if(devuelto)return; devuelto=true;
    /* una vez devuelta la frase no hay nada más que vigilar */
    vigias.forEach(v=>v.disconnect());
    document.querySelectorAll(PARTIDOS).forEach(el=>{
      const t=original.get(el);
      /* un nodo de texto nuevo: el traductor observa el DOM y lo recoge */
      if(t)el.replaceChildren(document.createTextNode(t));
    });
  }

  /* Chrome y Google marcan la raíz al traducir: le ponen translated-ltr
     (o -rtl). Basta con vigilar esa clase, que es un solo atributo de un
     solo elemento; observar todo el árbol para cazar los <font> con que
     envuelven lo traducido costaba fotogramas en cada scroll. */
  /* Si el selector de idioma de la página ya ha traducido, esto no se mete:
     ese sistema recompone los titulares por su cuenta y devolverlos al inglés
     desde aquí sería pisarle la traducción.
     Quién manda lo dice el propio botón del selector, que arriba queda con
     translate="no" y por tanto nadie más reescribe. No vale html.lang, porque
     el traductor de Chrome también lo reescribe y esto se apagaría justo
     cuando hace falta; ni localStorage, porque la página abre siempre en
     inglés a propósito y no guarda nada. */
  function propio(){
    const b=document.querySelector('.lang');
    return (!b || /^English/.test(b.textContent.trim())) ? 'en' : 'otro';
  }
  /* Chrome y Google marcan la raíz al traducir: le ponen translated-ltr o -rtl */
  function traducida(){
    if(propio()!=='en')return false;
    return /\btranslated-(ltr|rtl)\b/.test(document.documentElement.className)
        || !!document.querySelector('.goog-te-banner-frame,#goog-gt-tt');
  }

  guardar();
  addEventListener('load',guardar);

  if(traducida())entero();
  const raiz=new MutationObserver(()=>{ if(traducida())entero(); });
  raiz.observe(document.documentElement,{attributes:true,attributeFilter:['class','lang']});
  vigias.push(raiz);
  /* Y por si el traductor empieza por el texto antes de marcar la raíz: se
     vigila la aparición de los <font> con que envuelve lo que traduce, pero
     solo dentro de los titulares partidos —dieciocho elementos que por su
     cuenta no cambian nunca—, y no en todo el documento, que costaba
     fotogramas en cada scroll. */
  document.querySelectorAll(PARTIDOS).forEach(el=>{
    const v=new MutationObserver(ms=>{
      for(const m of ms)for(const n of m.addedNodes)
        if(n.nodeType===1&&n.tagName==='FONT'&&propio()==='en'){entero();return}
    });
    v.observe(el,{childList:true,subtree:true});
    vigias.push(v);
  });
  /* red de seguridad por si algún traductor no hace ninguna de las dos cosas:
     mirar una cadena cada dos segundos no le cuesta nada al navegador */
  const reloj=setInterval(()=>{ noTraducir(); if(traducida())entero(); },2000);
  vigias.push({disconnect:()=>clearInterval(reloj)});
})();
</script>
